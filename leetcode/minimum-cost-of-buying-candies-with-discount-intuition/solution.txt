class Solution {
public:
    int minimumCost(vector<int>& cost) {
        sort(cost.rbegin() , cost.rend());
        // 2  2  5  6  7  9
        long long ans = 0 , cnt = 1;
        for(int i=0; i<cost.size(); i++){
            if(cnt % 3 == 0){
                cnt = 1;
                continue;
            } 
            ans += cost[i];
            cnt++;
