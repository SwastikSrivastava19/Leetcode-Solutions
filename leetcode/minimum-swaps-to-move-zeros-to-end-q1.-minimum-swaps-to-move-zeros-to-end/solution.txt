// Problem: Q1. Minimum Swaps to Move Zeros to End

class Solution {public:    int minimumSwaps(vector<int>& nums) {        int nzero = 0;        for(int x : nums){            if(x != 0) nzero++;        }        int ans = 0;        for(int i=0; i<nzero; i++){            if(nums[i] == 0) ans++;        }        return ans;    }};
